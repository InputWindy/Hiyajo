FArchive::operator<<(std::string& V)

← 公开 API · FArchive&

operator<<(V): 1. Len = (uint32)V.size() 2. *this << Len // 长度前缀 3. if IsReading(): V.resize(Len) // 读时先预留 4. if Len > 0: Serialize(V.data(), Len) 5. return *this

FMemoryReader::Serialize(void* Out, size_t Size)

← 公开 API · void

Serialize(Out, Size): 1. if Pos + Size > Data.size(): return // 越界暂跳过 2. memcpy(Out, Data.data() + Pos, Size) 3. Pos += Size

FMemoryReader::Seek / FMemoryWriter::Seek

← 公开 API · void

Reader.Seek(Pos): 1. Pos = InPos Writer.Seek(Pos): 1. if InPos < Buffer.size(): Buffer.resize(InPos) // 回退则截断

FMemoryReader::Tell / FMemoryWriter::Tell

← 公开 API · size_t

Reader.Tell(): 1. return Pos Writer.Tell(): 1. return Buffer.size() // 缓冲大小即写入位置

FMemoryWriter::Serialize(void* In, size_t Size)

← 公开 API · void

Serialize(In, Size): 1. Old = Buffer.size() 2. Buffer.resize(Old + Size) 3. memcpy(Buffer.data() + Old, In, Size)

FMemoryWriter::GetBytes / TakeBytes

← 公开 API
