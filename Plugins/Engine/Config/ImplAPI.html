bool FConfig::Load(std::string_view Path)

← 公开 API · bool

INI 解析：逐行读——Trim 后跳过空行与 ';'/'#' 注释；"[Section]" 切换当前节；"key=value" 行写入 Sections[节][key]=value。文件打不开返回 false。

Load(Path): 1. Stream = ifstream(Path); if !Stream: return false 2. Section = "" 3. for 每行: 4. T = Trim(行) 5. if 空 或 T[0] ∈ {';','#'}: continue 6. if T 形如 "[...]": Section = 去括号; continue 7. Eq = T.find('=') 8. if Eq == npos: continue 9. Sections[Section][Trim(key)] = Trim(value) 10. return true

std::string Trim(std::string_view S)

内部 · std::string

去掉首尾空白（空格 / 制表 / 回车）。

Trim(S): 1. First = find_first_not_of(" \t\r") 2. if npos: return "" 3. Last = find_last_not_of(" \t\r") 4. return S[First..Last]

std::optional<std::string> FConfig::GetString(std::string_view Section, std::string_view Key) const

← 公开 API · std::optional<std::string>

取原始字符串值——节或 key 缺失返回 nullopt。

GetString(Section, Key): 1. SecIt = Sections.find(Section); if end: return nullopt 2. KeyIt = SecIt->second.find(Key); if end: return nullopt 3. return KeyIt->second

std::int64_t FConfig::GetInt(std::string_view Section, std::string_view Key, std::int64_t Default) const

← 公开 API · std::int64_t

整数读取——std::stoll 解析，缺失或解析失败回退 Default。

GetInt(Section, Key, Default): 1. V = GetString(Section, Key); if !V: return Default 2. try return stoll(*V); catch(...) return Default

← 公开 API · double

GetFloat(Section, Key, Default): 1. V = GetString(Section, Key); if !V: return Default 2. try return stod(*V); catch(...) return Default

bool FConfig::GetBool(std::string_view Section, std::string_view Key, bool Default) const

← 公开 API · bool

布尔读取——值转小写后与真值集 {"true","1","yes","on"} 比较，缺失回退 Default。

GetBool(Section, Key, Default): 1. V = GetString(Section, Key); if !V: return Default 2. Lower = 小写(*V) 3. return Lower ∈ {"true","1","yes","on"}

← 公开 API · void

SetString(Section, Key, Value): 1. Sections[Section][Key] = move(Value)
